---

```yaml
services:
  database:
    image: postgres
    volumes:
      - database-data:/var/lib/postgresql
    networks:
      - my-net

  backend:
    image: example/backend
    build:
      context: backend # docker build ./backend
    depends_on:
      - database
    networks:
      - my-net

#Eigentlich unnötig, per default sind alle im selben network
networks:
  my-net:

volumes:
  database-data:
```

#### Environment

```yaml
services:
  database:
    image: postgres
    ports:
      - 5432:5432
    environment:
      - POSTGRES_DB= ${POSTGRES_DB} # get value from environment
      - POSTGRES_USER= ${POSTGRES_DB:-postgres} # default postgres
    secrets:
      - db_password

secrets:
  db_password:
    file: ./secrets/db_password.txt # /run/secrets/db_password

```
