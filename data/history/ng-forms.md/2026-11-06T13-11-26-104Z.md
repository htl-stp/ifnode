#### Validation
```typescript
const loginForm = form(this.deployModel, (model) => {
  required(model.name);
  email(model.email);
  maxLength(model.name, 100);
  min(model.port, 1, {message: 'Port must be positive'});
  max(model.port, 65535);
  validate(model.port, ({value}) => {
    const port = value();
    return [0, 443, 5432].includes(port)
      ? {kind: 'forbidden-port', message: 'Forbidden port'}
      : null;
  });
  validate(formModel.envVars, ({value}) => {
    const envVars = value();
    const names = envVars.map((envVar) => envVar.name);
    return new Set(names).size !== names.length
      ? {kind: 'duplicate-env-vars', message: 'Duplicate Env vars'}
      : null;
  });
});
```

addEnvVar() {
  this.deployModel.update((current) => ({
    ...current,
    envVars: [...current.envVars, {name: '', value: ''}]
  }));
}
